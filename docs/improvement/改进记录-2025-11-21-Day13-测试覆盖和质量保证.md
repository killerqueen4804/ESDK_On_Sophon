/**
 * @brief 测试2: buildEvent() 算法分组
 *
 * 验证目标:
 * - 检测框按 algorithm 字段正确分组
 * - 每个算法生成独立的事件
 * - Base64 编码正确

void test2_BuildEvent_AlgorithmGrouping() {
    printTestHeader("测试2: buildEvent() 算法分组逻辑");

    // 准备测试数据: 4 个检测框，3 种算法
    std::vector<types::Detection> detections = {
        {"person", 0.95f, 100.0, 100.0, 50.0, 80.0},    // yolov5
        {"car", 0.88f, 200.0, 150.0, 80.0, 60.0},       // yolov5
        {"person", 0.92f, 300.0, 120.0, 45.0, 75.0},    // yolov8
        {"truck", 0.85f, 400.0, 200.0, 100.0, 70.0}     // faster_rcnn
    };

    // 模拟算法分组
    std::map<std::string, std::vector<types::Detection>> grouped;
    std::vector<std::string> algorithms = {"yolov5", "yolov5", "yolov8", "faster_rcnn"};

    for (size_t i = 0; i < detections.size(); i++) {
        grouped[algorithms[i]].push_back(detections[i]);
    }

    // 验证分组结果
    assert(grouped.size() == 3);                      // 应该有 3 组
    assert(grouped["yolov5"].size() == 2);           // yolov5 有 2 个
    assert(grouped["yolov8"].size() == 1);           // yolov8 有 1 个
    assert(grouped["faster_rcnn"].size() == 1);      // faster_rcnn 有 1 个

    std::cout << "分组结果:\n";
    for (const auto& [algo, dets] : grouped) {
        std::cout << "  " << algo << ": " << dets.size() << " 个检测框\n";
    }

    printTestResult(true, "算法分组逻辑正确");

/**
 * @brief 测试3: EventCache::publishEvent() 集成
 *
 * 验证目标:
 * - MediaFileTask 使用 EventCache 推送事件
 * - 不再使用 logger.debug() 记录
 * - 缓存和重试机制正常工作
 */
void test3_EventCache_Integration() {
    printTestHeader("测试3: EventCache 集成验证");

    auto& eventCache = EventCache::getInstance();

    // 清空缓存
    eventCache.clearAll();

    // 设置 Mock 发布回调（模拟 MQTT 断线）
    bool mqttConnected = false;
    eventCache.setPublishCallback([&mqttConnected](const std::string& topic,
                                                    const std::string& payload) {
        if (mqttConnected) {
            std::cout << "  [Mock MQTT] 发布成功: " << topic << "\n";
            return true;
        } else {
            std::cout << "  [Mock MQTT] 发布失败（断线），已缓存\n";
            return false;
        }
    });

    // 场景1: MQTT 断线，发布 2 个事件
    nlohmann::json event1 = {{"taskID", 1001}, {"test", "event1"}};
    nlohmann::json event2 = {{"taskID", 1002}, {"test", "event2"}};

    bool result1 = eventCache.publishEvent("test/topic", event1);
    bool result2 = eventCache.publishEvent("test/topic", event2);

    std::cout << "断线场景: 缓存事件数 = " << eventCache.getCachedEventCount() << "\n";
    assert(eventCache.getCachedEventCount() == 2);  // 应该缓存了 2 个

    // 场景2: MQTT 重连，自动重试
    mqttConnected = true;
    eventCache.onMqttReconnected();

    std::cout << "重连后: 缓存事件数 = " << eventCache.getCachedEventCount() << "\n";
    assert(eventCache.getCachedEventCount() == 0);  // 应该清空

    printTestResult(true, "EventCache 集成正常");
}

## 📚 核心知识点总结

### 1️⃣ 单元测试 vs 集成测试

#### 定义对比

| 对比项       | 单元测试 (Unit Test) | 集成测试 (Integration Test) |
| ------------ | -------------------- | --------------------------- |
| **测试范围** | 单个函数/类          | 多个模块组合                |
| **依赖处理** | 使用 Mock/Stub 隔离  | 使用真实依赖                |
| **运行速度** | 快（毫秒级）         | 慢（秒级）                  |
| **测试目的** | 验证单元逻辑正确     | 验证模块协作正确            |
| **失败原因** | 单元逻辑错误         | 接口不匹配、数据流错误      |

class Processor {
private:

public:

3. **单一职责** - 每个类/函数只做一件事
4. **避免全局状态** - 减少隐式依赖

### 2️⃣ Mock 对象设计

#### 什么是 Mock？

**Mock** 是测试中使用的"假对象"，用于替代真实的依赖，使测试：

- **快速** - 不需要真实的硬件/网络
- **稳定** - 不受外部环境影响
- **可控** - 可以模拟各种场景（成功、失败、超时等）

#### 项目中的 Mock 示例

**Mock MQTT 客户端**:

```cpp
/**
 * @brief Mock MQTT 发布回调
 *
 * 用于测试 EventCache 的缓存和重试机制，
 * 可以控制发布成功或失败。
 */
class MockMqttPublisher {
public:
    bool shouldSucceed = true;        // 控制发布是否成功
    int publishCount = 0;             // 记录发布次数
    std::vector<std::string> topics;  // 记录发布的主题

    bool publish(const std::string& topic, const std::string& payload) {
        publishCount++;
        topics.push_back(topic);

        std::cout << "  [Mock MQTT] 发布事件: topic=" << topic
                  << ", result=" << (shouldSucceed ? "成功" : "失败") << "\n";

        return shouldSucceed;  // 返回可控的结果
    }

    void reset() {
        publishCount = 0;
        topics.clear();
        shouldSucceed = true;
    }
};

// 使用 Mock
MockMqttPublisher mockMqtt;
mockMqtt.shouldSucceed = false;  // 模拟 MQTT 断线

EventCache& cache = EventCache::getInstance();
cache.setPublishCallback([&mockMqtt](const std::string& topic,
                                     const std::string& payload) {
    return mockMqtt.publish(topic, payload);
});

// 测试缓存逻辑
cache.publishEvent("test/topic", event);  // 应该缓存
assert(cache.getCachedEventCount() == 1); // 验证缓存

// 模拟重连
mockMqtt.shouldSucceed = true;
cache.onMqttReconnected();
assert(mockMqtt.publishCount == 1);       // 验证重试
```

#### Mock 设计原则

1. **记录调用** - 记录被调用的次数和参数

   ```cpp
   int callCount = 0;
   std::vector<std::string> arguments;
   ```

2. **可控返回值** - 可以控制返回成功/失败

   ```cpp
   bool shouldSucceed = true;
   bool returnValue = shouldSucceed ? true : false;
   ```

3. **行为验证** - 验证被调用的次数、顺序、参数

   ```cpp
   assert(mockMqtt.publishCount == 3);
   assert(mockMqtt.topics[0] == "test/topic");
   ```

4. **状态重置** - 提供 reset() 方法清理状态
   ```cpp
   void reset() {
       callCount = 0;
       arguments.clear();
   }
   ```

#### Mock vs Stub vs Fake

| 类型     | 定义         | 用途             | 示例                        |
| -------- | ------------ | ---------------- | --------------------------- |
| **Mock** | 验证交互行为 | 测试是否正确调用 | MockMqtt 记录调用次数       |
| **Stub** | 提供固定响应 | 替换真实数据源   | StubDatabase 返回固定数据   |
| **Fake** | 简化的实现   | 替换复杂依赖     | FakeFileSystem 内存文件系统 |

**项目示例**:

```cpp
// Mock: 验证 MQTT 是否被正确调用
class MockMqtt {
    int publishCount = 0;  // 记录调用
    bool publish(...) {
        publishCount++;    // 验证行为
        return true;
    }
};

// Stub: 提供固定的检测结果
class StubVision {
    std::vector<Detection> detect(...) {
        // 返回固定的检测结果，不实际运行检测器
        return {{"person", 0.95, 100, 100, 50, 80}};
    }
};

// Fake: 简化的文件系统
class FakeFileSystem {
    std::map<std::string, std::string> files_;  // 内存存储

    bool exists(const std::string& path) {
        return files_.count(path) > 0;
    }

    std::string read(const std::string& path) {
        return files_[path];
    }
};
```

#### 面试要点

**Q: 什么时候用 Mock，什么时候用真实对象？**

A:

```
使用 Mock:
✅ 外部依赖（网络、硬件、数据库）
✅ 慢速操作（TPU 推理、大文件 I/O）
✅ 不稳定的依赖（第三方 API）
✅ 需要模拟特定场景（错误、超时）

使用真实对象:
✅ 纯函数（Base64 编码、字符串处理）
✅ 轻量级对象（容器、结构体）
✅ 测试重点（被测试的核心逻辑）
✅ 集成测试（验证模块协作）
```

**Q: 如何避免 Mock 导致的"假阳性"测试？**

A: "假阳性"指测试通过但代码有 bug，因为 Mock 行为与真实对象不一致。

**避免方法**:

1. **保持 Mock 简单** - 只 mock 必要的行为
2. **参考真实接口** - Mock 接口与真实接口一致
3. **补充集成测试** - 用真实对象验证
4. **契约测试** - 验证 Mock 行为符合接口契约

```cpp
// ❌ 危险的 Mock（行为不真实）
class MockVision {
    std::vector<Detection> detect(...) {
        // 总是返回成功，但真实 Vision 可能失败
        return {{"person", 0.95, 100, 100, 50, 80}};
    }
};

// ✅ 更真实的 Mock
class MockVision {
    bool shouldFail = false;  // 可以模拟失败

    std::vector<Detection> detect(...) {
        if (shouldFail) {
            throw std::runtime_error("TPU error");  // 模拟真实错误
        }
        return {{"person", 0.95, 100, 100, 50, 80}};
    }
};
```

### 3️⃣ 测试驱动开发 (TDD)

#### TDD 的三个步骤

```
1. ❌ Red   - 写失败的测试
   ↓
2. ✅ Green - 写最少代码使测试通过
   ↓
3. 🔄 Refactor - 重构代码，保持测试通过
   ↓
   循环
```

#### 项目中的 TDD 实践

**场景**: 实现算法分组功能

**步骤 1: Red（写失败的测试）**

```cpp
// test_algorithm_grouping.cpp
void test_AlgorithmGrouping() {
    std::vector<Detection> detections = {
        {"person", 0.95, 100, 100, 50, 80},  // yolov5
        {"car", 0.88, 200, 150, 80, 60}      // yolov5

}

// 运行测试 → ❌ 编译失败（函数不存在）
```

**步骤 2: Green（写最少代码）**

```cpp
// MediaFileTask.cpp
std::map<std::string, std::vector<Detection>>
groupByAlgorithm(const std::vector<Detection>& detections) {
    std::map<std::string, std::vector<Detection>> grouped;

    for (const auto& det : detections) {
        // 假设每个检测框有 algorithm 字段
        grouped[det.algorithm].push_back(det);
    }

    return grouped;
}

// 运行测试 → ✅ 通过！
```

**步骤 3: Refactor（重构优化）**

```cpp
// 重构：添加错误处理和日志
std::map<std::string, std::vector<Detection>>
groupByAlgorithm(const std::vector<Detection>& detections) {
    std::map<std::string, std::vector<Detection>> grouped;

    if (detections.empty()) {
        logger_.warning("检测列表为空，无法分组");
        return grouped;
    }

    for (const auto& det : detections) {
        if (det.algorithm.empty()) {
            logger_.warning("检测框缺少 algorithm 字段，跳过");
            continue;
        }
        grouped[det.algorithm].push_back(det);
    }

    logger_.debug("分组完成: {} 种算法", grouped.size());
    return grouped;
}

// 运行测试 → ✅ 依然通过！
```

#### TDD 的优势

1. **设计先行** - 先思考接口，再实现
2. **快速反馈** - 立即知道代码是否正确
3. **回归保护** - 重构时测试保证功能不变
4. **文档作用** - 测试即文档，展示如何使用

#### TDD 的局限

1. **学习曲线** - 需要时间适应思维转变
2. **初期慢** - 前期写测试增加时间
3. **不适用所有场景** - 探索性开发、UI 开发不适合

**Q: Red-Green-Refactor 循环的意义是什么？**

A:

```

  小步迭代 → 快速反馈 → 持续改进
  类似敏捷开发的短迭代思想
```

### 4️⃣ 测试覆盖率

#### 什么是测试覆盖率？

**测试覆盖率** 衡量测试执行了多少代码。

常见指标：

- **行覆盖率** (Line Coverage) - 执行的代码行数占比
- **分支覆盖率** (Branch Coverage) - 执行的条件分支占比
- **函数覆盖率** (Function Coverage) - 执行的函数占比

## 🎓 面试重点回顾

### 1. 测试策略

**问题**: "你如何为一个新功能设计测试？"

**回答模板**:

```
1. 分析功能范围
   - 核心逻辑是什么？
   - 依赖哪些模块？
   - 有哪些边界情况？

2. 制定测试策略
   - 单元测试: 核心算法、数据处理
   - 集成测试: 模块协作、数据流转
   - E2E 测试: 真实场景、用户体验

3. 实施测试
   - 先写核心逻辑测试
   - 再写边界情况测试
   - 最后写集成测试

4. 验证和优化
   - 检查覆盖率
   - 优化测试性能
   - 建立 CI 流程

项目示例: Day 11/12 功能测试...
```

### 2. 代码质量

**问题**: "如何保证代码质量？"

**回答要点**:

- **自动化测试** - 单元测试、集成测试、E2E 测试
- **代码审查** - Pull Request、结对编程
- **静态分析** - clang-tidy、cppcheck
- **持续集成** - 每次提交都运行测试
- **性能测试** - 建立性能基准，防止退化

### 3. TDD 实践

**问题**: "什么是 TDD？你实践过吗？"

**回答模板**:

```
TDD 是测试驱动开发，遵循 Red-Green-Refactor 循环。

我在项目中用 TDD 开发算法分组功能：
1. Red: 先写测试 test_AlgorithmGrouping()
2. Green: 实现 groupByAlgorithm() 让测试通过
3. Refactor: 优化代码，添加错误处理

好处:
  - 设计先行，接口更清晰
  - 测试覆盖率高
  - 重构有保障

挑战:
  - 初期需要时间适应
  - 不适合探索性开发

我的经验: 核心逻辑用 TDD，快速原型先实现再补测试
```
