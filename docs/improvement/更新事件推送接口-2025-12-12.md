```cpp
struct DetectionEvent {
    // GPS 位置信息
    struct GpsLocation {
        double lon;
        double lat;
    };

    // 掩码轮廓点
    struct MaskPoint {
        int x, y;
        GpsLocation location;
    };

    // 检测框
    struct BBox {
        int x, y, w, h;
        GpsLocation location;  // 中心点 GPS
    };

    // 单个检测对象
    struct DetectedObject {
        std::string label;
        BBox bbox;
        std::vector<MaskPoint> mask;  // 可选
    };

    // 顶层字段
    std::string uuid;
    int taskID;           // 改为 int
    int eventType;
    int main_type;        // 字段名改为 main_type
    std::string createTime;
    std::string eventDescribe;
    double longitude, latitude;
    std::string fileName;

    // result 字段
    std::string resultImage;           // Base64
    std::vector<DetectedObject> objects;
};
```

### 3. 实现 eventToJson()

生成嵌套的 JSON 结构:

```cpp
std::string TaskService::eventToJson(const DetectionEvent& event) {
    nlohmann::json j;

    // 顶层字段
    j["UUID"] = event.uuid;
    j["taskID"] = event.taskID;
    j["eventType"] = event.eventType;
    j["main_type"] = event.main_type;
    j["createTime"] = event.createTime;
    j["eventDescribe"] = event.eventDescribe;
    j["longitude"] = event.longitude;
    j["latitude"] = event.latitude;
    j["fileName"] = event.fileName;

    // result 字段
    nlohmann::json result;
    result["image"] = event.resultImage;

    // objects 数组
    nlohmann::json objects = nlohmann::json::array();
    for (const auto& obj : event.objects) {
        nlohmann::json objJson;
        objJson["label"] = obj.label;

        // bbox
        nlohmann::json bbox;
        bbox["x"] = obj.bbox.x;
        bbox["y"] = obj.bbox.y;
        bbox["w"] = obj.bbox.w;
        bbox["h"] = obj.bbox.h;
        bbox["location"]["lon"] = obj.bbox.location.lon;
        bbox["location"]["lat"] = obj.bbox.location.lat;
        objJson["bbox"] = bbox;

        // mask (可选)
        if (!obj.mask.empty()) {
            nlohmann::json mask = nlohmann::json::array();
            for (const auto& pt : obj.mask) {
                nlohmann::json point;
                point["x"] = pt.x;
                point["y"] = pt.y;
                point["location"]["lon"] = pt.location.lon;
                point["location"]["lat"] = pt.location.lat;
                mask.push_back(point);
            }
            objJson["mask"] = mask;
        }

        objects.push_back(objJson);
    }
    result["objects"] = objects;

    j["result"] = result;

    return j.dump();
}
```

#### 6. 添加分割可视化保存 (~Line 499-550)

**功能**: 在 `processSegmentation()` 中保存可视化图片到本地

**代码实现**:

```cpp
// 6. 保存分割可视化图片到本地
try {
    // 生成可视化图片
    cv::Mat visFrame = frame.clone();

    for (const auto& result : segResults) {
        // 绘制检测框 (绿色)
        cv::Rect rect(result.box.x, result.box.y, result.box.width, result.box.height);
        cv::rectangle(visFrame, rect, cv::Scalar(0, 255, 0), 2);

        // 绘制分割轮廓 (蓝色)
        if (!result.contours.empty()) {
            std::vector<std::vector<cv::Point>> contours = {result.contours};
            cv::drawContours(visFrame, contours, 0, cv::Scalar(255, 0, 0), 2);
        } else if (!result.mask.empty()) {
            std::vector<std::vector<cv::Point>> contours;
            cv::findContours(result.mask, contours, cv::RETR_EXTERNAL, cv::CHAIN_APPROX_SIMPLE);
            if (!contours.empty()) {
                cv::drawContours(visFrame, contours, -1, cv::Scalar(255, 0, 0), 2);
            }
        }

        // 绘制类别标签 (绿色文字)
        std::string label = result.className + " (" + std::to_string((int)(result.confidence * 100)) + "%)";
        cv::putText(visFrame, label,
                   cv::Point(result.box.x, result.box.y - 10),
                   cv::FONT_HERSHEY_SIMPLEX, 0.6, cv::Scalar(0, 255, 0), 2);
    }

    // 保存到 /data/Edge-SDK/results/seg_vis/
    std::string outputDir = "/data/Edge-SDK/results/seg_vis/";

    // 确保目录存在
    struct stat info;
    if (stat(outputDir.c_str(), &info) != 0) {
        mkdir(outputDir.c_str(), 0755);  // POSIX
    }

    // 文件名: seg_<taskID>_<timestamp>.jpg
    std::string outputPath = outputDir + "seg_" + std::to_string(config.taskID) +
                             "_" + getCurrentTimeString() + ".jpg";

    if (cv::imwrite(outputPath, visFrame)) {
        impl_->logger.info("🎨 [分割可视化] 已保存: " + outputPath);
    }

} catch (const std::exception& e) {
    impl_->logger.error("❌ [分割可视化] 保存异常: " + std::string(e.what()));
    // 可视化保存失败不影响主流程
}
```

**设计要点**:

- ✅ **颜色区分**: 检测框(绿色) vs 分割轮廓(蓝色)
- ✅ **置信度标签**: 显示类别名和置信度百分比
- ✅ **时间戳文件名**: 避免重名覆盖
- ✅ **异常安全**: 保存失败不影响主流程
- ✅ **目录自动创建**: 使用 POSIX mkdir

**文件命名规则**:

```
/data/Edge-SDK/results/seg_vis/seg_<taskID>_<YYYYMMDD_HHMMSS_mmm>.jpg

示例:
seg_12345_20251212_143025_567.jpg
```

**为什么要单独保存可视化图片?**

1. **调试验证**: 检查 SAM2 分割质量是否符合预期
2. **问题排查**: 如果云端显示异常,可以对比本地图片
3. **离线分析**: 后续可以回溯查看历史分割结果
4. **性能监控**: 观察不同场景下的分割效果
