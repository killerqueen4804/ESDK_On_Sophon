# ============================================================================
# src/core/CMakeLists.txt - 核心模块配置
# ============================================================================
# 功能：提供日志系统、配置管理等基础设施
# 依赖：无（仅依赖标准库和Qt5）
# 被依赖：几乎所有模块都依赖core
# ============================================================================

# ----------------------------------------------------------------------------
# 1. 定义源文件列表
# ----------------------------------------------------------------------------
# ⚠️ 注意：Application.cpp 在 src/ 目录，不属于 core 模块
# core 模块只包含基础设施组件：日志、配置、缓存等
set(CORE_SOURCES
    Logger.cpp
    Config.cpp
    EventCache.cpp  # MQTT 事件缓存与重试
)

# ----------------------------------------------------------------------------
# 2. 创建静态库
# ----------------------------------------------------------------------------
# STATIC：编译为静态库 libcore.a
# 为什么用静态库？
# 1. 便于增量编译：只修改core时，只重新编译libcore.a
# 2. 模块化管理：清晰的依赖关系
# 3. 可复用：其他项目可以直接使用这个库
add_library(core STATIC
    ${CORE_SOURCES}
)

# 设置库的别名（推荐的现代CMake实践）
add_library(esdk_sophon::core ALIAS core)

# ----------------------------------------------------------------------------
# 3. 设置头文件包含路径
# ----------------------------------------------------------------------------
# PUBLIC：使用core库的其他目标也能看到这些头文件
# $<BUILD_INTERFACE:...>：编译时使用的路径
# $<INSTALL_INTERFACE:...>：安装后使用的路径（如果需要安装）
target_include_directories(core
    PUBLIC
        $<BUILD_INTERFACE:${CMAKE_SOURCE_DIR}/include>
        $<INSTALL_INTERFACE:include>
)

# ----------------------------------------------------------------------------
# 4. 链接第三方库
# ----------------------------------------------------------------------------
# PRIVATE：这些库只在core内部使用，不会暴露给依赖core的模块
target_link_libraries(core
    PRIVATE
        ThirdParty::Qt5Core      # Qt核心功能
        ThirdParty::Json         # JSON配置解析
)

# ----------------------------------------------------------------------------
# 5. 设置编译选项
# ----------------------------------------------------------------------------
# PRIVATE：这些选项只在编译core时使用
target_compile_options(core
    PRIVATE
        $<$<CXX_COMPILER_ID:GNU>:-Wall -Wextra>      # GCC警告
        $<$<CXX_COMPILER_ID:MSVC>:/W4>               # MSVC警告
)

# ----------------------------------------------------------------------------
# 6. 设置库输出路径
# ----------------------------------------------------------------------------
set_target_properties(core PROPERTIES
    ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib"
)

# ============================================================================
# 📚 知识点：静态库 vs 动态库
# ============================================================================
# 静态库（STATIC）：
# - 编译时链接到可执行文件中
# - 优点：无需额外的.so/.dll文件，部署简单
# - 缺点：可执行文件体积较大
# - Linux: libcore.a    Windows: core.lib
#
# 动态库（SHARED）：
# - 运行时加载
# - 优点：多个程序可共享，节省内存
# - 缺点：需要额外的.so/.dll文件
# - Linux: libcore.so   Windows: core.dll
#
# 本项目选择STATIC的原因：
# 1. 嵌入式设备，简化部署
# 2. 不需要多程序共享
# 3. 避免动态库版本冲突
# ============================================================================

# ============================================================================
# 📚 知识点：PUBLIC vs PRIVATE vs INTERFACE
# ============================================================================
# 
# target_include_directories的可见性：
# 
# PUBLIC：
#   - 编译core时需要
#   - 链接core的目标也需要
#   - 例：头文件路径，mqtt链接core后自动获得头文件路径
#
# PRIVATE：
#   - 只在编译core时需要
#   - 链接core的目标看不到
#   - 例：内部实现用的第三方库
#
# INTERFACE：
#   - 编译core时不需要
#   - 链接core的目标需要
#   - 例：header-only库（只有头文件）
#
# 面试常考场景：
# Q: 为什么include路径用PUBLIC，而Qt5Core用PRIVATE？
# A: 因为mqtt等模块需要包含core的头文件（如Logger.h），所以include
#    路径要PUBLIC传递。但Qt5Core只在core内部实现中使用，mqtt不需要
#    直接使用Qt，所以用PRIVATE。
# ============================================================================
