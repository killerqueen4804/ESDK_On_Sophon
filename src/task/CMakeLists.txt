# ============================================================================
# src/task/CMakeLists.txt - 任务管理模块配置文件
# ============================================================================
# 功能：编译任务管理模块，管理算法任务的生命周期、TPU资源分配、并发控制
# ============================================================================

# ----------------------------------------------------------------------------
# 1. 收集源文件
# ----------------------------------------------------------------------------
set(TASK_SOURCES
    TaskManager.cpp
    TaskService.cpp     # Day 2: 业务逻辑编排层
    TaskTypes.cpp       # Day 3: 任务类型定义和 JSON 解析
    LiveStreamTask.cpp  # Day 4: 直播流任务实现
    MediaFileTask.cpp   # Day 4: 媒体文件任务实现
)

# ----------------------------------------------------------------------------
# 2. 创建静态库
# ----------------------------------------------------------------------------
add_library(task STATIC ${TASK_SOURCES})

# 设置库的别名（推荐的现代CMake实践）
add_library(esdk_sophon::task ALIAS task)

# ----------------------------------------------------------------------------
# 3. 设置头文件包含路径
# ----------------------------------------------------------------------------
# PUBLIC：依赖task的其他模块也能访问这些头文件
target_include_directories(task
    PUBLIC
        ${PROJECT_SOURCE_DIR}/include  # esdk_sophon/task/TaskManager.h
)

# ----------------------------------------------------------------------------
# 4. 链接依赖
# ----------------------------------------------------------------------------
target_link_libraries(task
    PUBLIC
        core             # Logger、配置管理
        utils            # ⭐ 改为PUBLIC：MediaFileTask.h中引用了ImageProcessor等工具类
        ThirdParty::Json # JSON解析（nlohmann/json）
        ThirdParty::OpenCV  # 图像编码（imencode）
        ThirdParty::EdgeSDK # DJI Edge SDK (Liveview, MediaManager) - Day 6
    PRIVATE
        mqtt             # MQTT客户端（TaskService需要）
        vision           # 视觉检测（TaskService需要）
        video            # H264解码器（LiveStreamTask需要） - Day 7
        rtmp             # RTMP推流器（LiveStreamTask需要） - Day 6
)

# ----------------------------------------------------------------------------
# 5. 设置编译选项
# ----------------------------------------------------------------------------
target_compile_features(task PUBLIC cxx_std_17)  # 使用C++17

# 添加编译警告
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(task PRIVATE
        -Wall          # 所有警告
        -Wextra        # 额外警告
        -Wpedantic     # 严格标准警告
        -Wno-unused-parameter  # 忽略未使用参数警告（接口预留参数）
    )
endif()

# ----------------------------------------------------------------------------
# 6. 设置库输出路径
# ----------------------------------------------------------------------------
set_target_properties(task PROPERTIES
    ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib"
)

# ============================================================================
# 📚 知识点：任务管理模块设计要点
# ============================================================================
# 1. 单例模式：TaskManager 全局唯一实例
# 2. 线程安全：使用 std::mutex 保护共享状态
# 3. RAII：使用 unique_ptr 管理辅助对象生命周期
# 4. TPU资源管理：预估显存、检查可用资源、并发控制
# 5. 任务冲突检测：视频流互斥、图片并发
# 6. 等待队列：资源不足时排队，资源释放时自动启动
# ============================================================================
